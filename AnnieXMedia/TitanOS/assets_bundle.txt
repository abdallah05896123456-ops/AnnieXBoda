/**
 * ======================================================================================
 * TITAN OS | NEURAL ENGINE v10.0
 * MODULE: CORE LOGIC, WEBSOCKETS, VISUALS, & AUDIO
 * DEVELOPED BY: TITAN SYSTEMS
 * ======================================================================================
 */

// --- [1] GLOBAL SYSTEM CLASS ---
class TitanSystem {
    constructor() {
        this.socket = null;
        this.activeChatId = -100123456789; // Mock ID
        this.isConnected = false;
        
        // Modules
        this.particles = null;
        this.visualizer = null;
        
        // Config
        this.config = {
            wsUrl: (location.protocol === 'https:' ? 'wss' : 'ws') + '://' + location.host + '/ws/monitor',
            reconnectMs: 3000
        };

        this.init();
    }

    init() {
        console.log("%c TITAN OS %c SYSTEM STARTUP", "background: #000; color: #00f2ff; padding: 5px; border: 1px solid #00f2ff;", "color: #fff;");
        
        // 1. Start Subsystems
        this.initParticles();
        this.initVisualizer();
        this.setupWebSocket();
        this.bindEvents();
        
        // 2. Initial Mock Data (Immediate Feedback)
        this.updateStats({cpu: 25, ram: 40, net_sent: 1024, net_recv: 4096});
        this.notify("System Online", "TitanOS Kernel Loaded Successfully.", "success");
    }

    // --- [2] PARTICLE ENGINE ---
    initParticles() {
        const canvas = document.createElement('canvas');
        canvas.id = 'particle-canvas';
        document.body.prepend(canvas);
        const ctx = canvas.getContext('2d');
        
        let width, height;
        let particles = [];
        
        const resize = () => {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
        };
        window.addEventListener('resize', resize);
        resize();

        for(let i=0; i<60; i++) {
            particles.push({
                x: Math.random() * width, y: Math.random() * height,
                vx: (Math.random()-0.5)*0.5, vy: (Math.random()-0.5)*0.5,
                size: Math.random() * 2
            });
        }

        const animate = () => {
            ctx.clearRect(0,0,width,height);
            ctx.fillStyle = 'rgba(0, 242, 255, 0.4)';
            
            particles.forEach(p => {
                p.x += p.vx; p.y += p.vy;
                if(p.x < 0 || p.x > width) p.vx *= -1;
                if(p.y < 0 || p.y > height) p.vy *= -1;
                ctx.beginPath(); ctx.arc(p.x, p.y, p.size, 0, Math.PI*2); ctx.fill();
            });
            
            // Connect Lines
            ctx.strokeStyle = 'rgba(0, 242, 255, 0.04)';
            for(let i=0; i<particles.length; i++) {
                for(let j=i+1; j<particles.length; j++) {
                    const dx = particles[i].x - particles[j].x;
                    const dy = particles[i].y - particles[j].y;
                    if(dx*dx+dy*dy < 15000) {
                        ctx.beginPath(); ctx.moveTo(particles[i].x, particles[i].y);
                        ctx.lineTo(particles[j].x, particles[j].y); ctx.stroke();
                    }
                }
            }
            requestAnimationFrame(animate);
        };
        animate();
    }

    // --- [3] NETWORK LAYER ---
    setupWebSocket() {
        this.socket = new WebSocket(this.config.wsUrl);
        
        this.socket.onopen = () => {
            this.isConnected = true;
            document.querySelector('.status-dot').classList.add('online');
            this.notify("Network Link", "Secure WebSocket Connected.", "info");
        };

        this.socket.onmessage = (e) => {
            try { this.updateStats(JSON.parse(e.data)); } catch(err){}
        };

        this.socket.onclose = () => {
            this.isConnected = false;
            document.querySelector('.status-dot').classList.remove('online');
            this.notify("Connection Lost", "Retrying link...", "danger");
            setTimeout(() => this.setupWebSocket(), this.config.reconnectMs);
        };
    }

    // --- [4] UI & LOGIC ---
    updateStats(data) {
        // CPU
        const cpuElem = document.getElementById('cpu-live');
        if(cpuElem) {
            cpuElem.innerText = `${data.cpu}%`;
            document.querySelector('.cpu-fill').style.width = `${data.cpu}%`;
        }
        // RAM
        const ramElem = document.getElementById('ram-live');
        if(ramElem) {
            ramElem.innerText = `${data.ram}%`;
            document.querySelector('.ram-fill').style.width = `${data.ram}%`;
        }
    }

    // --- [5] AUDIO SYNTH (SOUND FX) ---
    playSfx(type) {
        try {
            const AudioContext = window.AudioContext || window.webkitAudioContext;
            if(!AudioContext) return;
            const ctx = new AudioContext();
            const osc = ctx.createOscillator();
            const gain = ctx.createGain();
            osc.connect(gain); gain.connect(ctx.destination);
            
            if(type === 'success') {
                osc.type = 'sine'; osc.frequency.setValueAtTime(600, ctx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(1000, ctx.currentTime + 0.1);
            } else if (type === 'hover') {
                osc.type = 'sine'; osc.frequency.setValueAtTime(400, ctx.currentTime);
                gain.gain.value = 0.05;
            } else { // Error/Click
                osc.type = 'triangle'; osc.frequency.setValueAtTime(150, ctx.currentTime);
            }
            
            if(type !== 'hover') gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.2);
            osc.start(); osc.stop(ctx.currentTime + (type==='hover'?0.05:0.2));
        } catch(e){}
    }

    // --- [6] NOTIFICATIONS (TOASTS) ---
    notify(title, msg, type='info') {
        const container = document.getElementById('toast-container') || this.createToastContainer();
        const toast = document.createElement('div');
        toast.className = 'toast';
        toast.style.borderLeftColor = type==='success'?'var(--neon-success)':(type==='danger'?'var(--neon-danger)':'var(--neon-cyan)');
        
        let icon = type==='success'?'fa-check-circle':(type==='danger'?'fa-triangle-exclamation':'fa-info-circle');
        
        toast.innerHTML = `
            <i class="fa-solid ${icon}" style="font-size:1.2rem; color:${toast.style.borderLeftColor}"></i>
            <div><h4 style="margin:0;font-size:0.9rem">${title}</h4><span style="font-size:0.8rem;color:#aaa">${msg}</span></div>
        `;
        
        container.appendChild(toast);
        this.playSfx(type);
        setTimeout(() => {
            toast.style.animation = 'slideInLeft 0.3s reverse forwards';
            setTimeout(() => toast.remove(), 300);
        }, 4000);
    }
    createToastContainer() {
        const d = document.createElement('div'); d.id='toast-container'; document.body.appendChild(d); return d;
    }

    // --- [7] VISUALIZER ---
    initVisualizer() {
        const canvas = document.getElementById('audio-visualizer');
        if(!canvas) return;
        const ctx = canvas.getContext('2d');
        
        const loop = () => {
            if(canvas.clientWidth !== canvas.width) canvas.width = canvas.clientWidth;
            if(canvas.clientHeight !== canvas.height) canvas.height = canvas.clientHeight;
            
            ctx.clearRect(0,0,canvas.width,canvas.height);
            const bars = 60; const barW = canvas.width/bars;
            
            for(let i=0; i<bars; i++) {
                // Simulate Audio Data
                const h = Math.random() * canvas.height * 0.8;
                ctx.fillStyle = `rgba(0, 242, 255, ${Math.random()})`;
                ctx.fillRect(i*barW, canvas.height-h, barW-2, h);
            }
            requestAnimationFrame(loop);
        };
        loop();
    }
    
    bindEvents() {
        // Sound on hover
        document.querySelectorAll('button, a').forEach(el => {
            el.addEventListener('mouseenter', () => this.playSfx('hover'));
        });
    }

    // API Bridge
    async apiCall(action) {
        this.notify("Command Sent", `Executing: ${action.toUpperCase()}`, "info");
        return fetch(`/api/${action}/${this.activeChatId}`, {method:'POST'}).catch(e=>console.log(e));
    }
}

// --- INITIALIZE ---
const Titan = new TitanSystem();

// --- EXPOSED FUNCTIONS (HTML BINDINGS) ---
window.togglePlay = () => {
    const btn = document.getElementById('play-btn-lg');
    const icon = btn.querySelector('i');
    if(icon.classList.contains('fa-play')) {
        icon.className = 'fa-solid fa-pause';
        document.querySelector('.card').classList.add('playing');
        Titan.apiCall('resume');
    } else {
        icon.className = 'fa-solid fa-play';
        document.querySelector('.card').classList.remove('playing');
        Titan.apiCall('pause');
    }
};
window.control = (act) => Titan.apiCall(act);
window.cleanCache = () => Titan.apiCall('cleancache');
window.activateTurbo = () => Titan.notify("TURBO MODE", "All systems overclocked!", "danger");
